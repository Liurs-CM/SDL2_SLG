74void setScaredTarget(entt::registry &reg, const MazeState &maze, std::mt19937 &rand) {
75 auto view = reg.view<Target, Position, ScaredMode, ActualDir>();
76 for (const entt::entity e : view) {
77 const Pos pos = view.get<Position>(e).p;
78 const Dir dir = view.get<ActualDir>(e).d;
79 const Pos nextPos = pos + toPos(dir);
80 std::uniform_int_distribution<int> dist(0, 3);
81 Dir candDir = static_cast<Dir>(dist(rand));
82 Pos candPos = nextPos;
83
84 for (int i = 0; i != 4; ++i) {
85 if (candDir == opposite(dir)) {
86 candDir = rotateCW(candDir);
87 continue;
88 }
89
90 if (!canMove(reg, maze, e, nextPos, candDir)) {
91 candDir = rotateCW(candDir);
92 continue;
93 }
94
95 candPos = nextPos + toPos(candDir);
96 break;
97 }
98
99 view.get<Target>(e).p = candPos;
100 }
101}
