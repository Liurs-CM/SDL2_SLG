21void pursueTarget(entt::registry &reg, const MazeState &maze) {
22 constexpr float inf = 1e10f;
23 auto view = reg.view<Target, Position, ActualDir, DesiredDir>();
24 for (const entt::entity e : view) {
25 const Pos targetPos = view.get<Target>(e).p;
26 // the position we are currently moving from
27 const Pos pos = view.get<Position>(e).p;
28 // the direction we are currently moving in
29 const Dir dir = view.get<ActualDir>(e).d;
30 // the position we are currently moving to
31 const Pos nextPos = pos + toPos(dir);
32 // distance from the 4 candiate directions to the target
33 float targetDist[4] = {inf, inf, inf, inf};
34 for (const Dir candDir : dir_range) {
35 // can't go back the way we came
36 if (candDir == opposite(dir)) {
37 continue;
38 }
39
40 // can't go through walls
41 if (!canMove(reg, maze, e, nextPos, candDir)) {
42 continue;
43 }
44
45 const Pos candPos = nextPos + toPos(candDir);
46 targetDist[static_cast<std::size_t>(candDir)] = distance(candPos, targetPos);
47 }
48
49 float *smallestDist = std::min_element(targetDist, targetDist + 4);
50 if (smallestDist == targetDist + 4 || *smallestDist == inf) {
51 view.get<DesiredDir>(e).d = dir;
52 continue;
53 }
54
55 view.get<DesiredDir>(e).d = static_cast<Dir>(smallestDist - targetDist);
56 }
57}
