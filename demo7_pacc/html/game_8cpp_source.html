40bool Game::logic() {
41 // The order of systems is very important in an ECS. Each system reads some
42 // state and modifies some state. If the state isn't read and modified in the
43 // right order, subtle bugs can occur. Make sure that the order of systems is
44 // easy to see (i.e. not hidden away by some abstraction that sets the order
45 // for you). Always think carefully about the order that systems should be in.
46
47 // It's OK to keep some game state outside of the ECS (e.g. maze, dots,
48 // dotSprite) but try to keep as much state within the ECS as you can.
49 // Keeping too much state outside of the ECS can lead to problems.
50 // For example: `dots` is the amount of dots eaten by the player. If there
51 // were more than one player, then each player might want to keep track of how
52 // many dots they've eaten. So `dots` would have to be moved into a component
53
54 if (state != State::playing) {
55 return true;
56 }
57
58 if (scattering) {
59 if (ticks >= scatterTicks) {
60 ghostChase(reg);
61 ticks = 0;
62 scattering = false;
63 }
64 } else {
65 if (ticks >= chaseTicks) {
66 ghostScatter(reg);
67 ticks = 0;
68 scattering = true;
69 }
70 }
71 ++ticks;
72
73 movement(reg);
74 wallCollide(reg, maze);
75 dots += eatDots(reg, maze);
76 if (eatEnergizer(reg, maze)) {
77 ghostScared(reg);
78 }
79 ghostScaredTimeout(reg);
80 enterHouse(reg);
81 setBlinkyChaseTarget(reg);
82 setPinkyChaseTarget(reg);
83 setInkyChaseTarget(reg);
84 setClydeChaseTarget(reg);
85 setScaredTarget(reg, maze, rand);
86 setScatterTarget(reg);
87 setEatenTarget(reg);
88 leaveHouse(reg);
89 pursueTarget(reg, maze);
90
91 const GhostCollision collision = playerGhostCollide(reg);
92 if (collision.type == GhostCollision::Type::eat) {
93 ghostEaten(reg, collision.ghost);
94 }
95 if (collision.type == GhostCollision::Type::lose) {
96 state = State::lost;
97 } else if (dots == dotsInMaze) {
98 state = State::won;
99 }
100 return true;
101}
