17void movement(entt::registry &reg) {
18 auto view = reg.view<Position, ActualDir>();
19 for (const entt::entity e : view) {
20 Pos &pos = view.get<Position>(e).p;
21 const Dir dir = view.get<ActualDir>(e).d;
22 pos += toPos(dir);
23
24 // The tunnel.
25 // This assumes the exact position of the tunnel.
26 // It's good enough for this simple game but a more robust solution might
27 // involve making the tunnel into an entity
28 if (pos.y == 10) {
29 if (pos.x <= -1 && dir == Dir::left) {
30 pos.x = 19;
31 } else if (pos.x >= 19 && dir == Dir::right) {
32 pos.x = -1;
33 }
34 }
35 }
36}

38void wallCollide(entt::registry &reg, const MazeState &maze) {
39 auto view = reg.view<Position, ActualDir, DesiredDir>();
40 for (const entt::entity e : view) {
41 const Pos pos = view.get<Position>(e).p;
42 const Dir desiredDir = view.get<DesiredDir>(e).d;
43 if (canMove(reg, maze, e, pos, desiredDir)) {
44 view.get<ActualDir>(e).d = desiredDir;
45 continue;
46 }
47
48 const Dir prevDir = view.get<ActualDir>(e).d;
49 if (canMove(reg, maze, e, pos, prevDir)) {
50 continue;
51 }
52
53 view.get<ActualDir>(e).d = Dir::none;
54 }
55}
