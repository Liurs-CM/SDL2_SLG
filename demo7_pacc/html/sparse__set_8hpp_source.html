165 page_type & assure(const std::size_t pos) {
166 if(!(pos < sparse.size())) {
167 sparse.resize(pos+1);
168 }
169
170 if(!sparse[pos]) {
171 sparse[pos] = std::make_unique<entity_type[]>(entt_per_page);
172 // null is safe in all cases for our purposes
173 for(auto *first = sparse[pos].get(), *last = first + entt_per_page; first != last; ++first) {
174 *first = null;
175 }
176 }
177
178 return sparse[pos];

555 template<typename Apply, typename Compare, typename Sort = std_sort, typename... Args>
556 void arrange(iterator first, iterator last, Apply apply, Compare compare, Sort algo = Sort{}, Args &&... args) {
557 ENTT_ASSERT(!(last < first));
558 ENTT_ASSERT(!(last > end()));
559
560 const auto length = std::distance(first, last);
561 const auto skip = std::distance(last, end());
562 const auto to = packed.rend() - skip;
563 const auto from = to - length;
564
565 algo(from, to, std::move(compare), std::forward<Args>(args)...);
566
567 for(size_type pos = skip, end = skip+length; pos < end; ++pos) {
568 auto curr = pos;
569 auto next = index(packed[curr]);
570
571 while(curr != next) {
572 apply(packed[curr], packed[next]);
573 sparse[page(packed[curr])][offset(packed[curr])] = entity_type(curr);
574
575 curr = next;
576 next = index(packed[curr]);
577 }
578 }

601 void respect(const sparse_set &other) {
602 const auto to = other.end();
603 auto from = other.begin();
604
605 size_type pos = packed.size() - 1;
606
607 while(pos && from != to) {
608 if(contains(*from)) {
609 if(*from != packed[pos]) {
610 swap(packed[pos], *from);
611 }
612
613 --pos;
614 }
615
616 ++from;
617 }
