128 const pool_handler<Event> & assure() const {
129 static_assert(std::is_same_v<Event, std::decay_t<Event>>);
130
131 if constexpr(has_type_index_v<Event>) {
132 const auto index = type_index<Event>::value();

135 pools.resize(index+1);
136 }

138 if(!pools[index]) {
139 pools[index].reset(new pool_handler<Event>{});

143 } else {

145 return static_cast<pool_handler<Event> &>(it == pools.cend() ? *pools.emplace_back(new pool_handler<Event>{}) : **it);
146 }
