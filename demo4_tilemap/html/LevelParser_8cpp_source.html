29Level* LevelParser::parseLevel(const char *levelFile)
30{
31 // create a tinyXML document and load the map xml
32 XMLDocument levelDocument;
33 if(levelDocument.LoadFile(levelFile) != XML_SUCCESS) {
34 std::cerr << "Failed to load level file: " << levelDocument.ErrorStr() << "\n";
35 return nullptr;
36 }
37
38 // get the root node and display some values
39 XMLElement* pRoot = levelDocument.RootElement();
40 std::cout << "Loading level:\n"
41 << "Version: " << getAttributeString(pRoot, "version") << "\n"
42 << "Width:" << getAttributeString(pRoot, "width")
43 << " - Height:" << getAttributeString(pRoot, "height") << "\n"
44 << "Tile Width:" << getAttributeString(pRoot, "tilewidth")
45 << " - Tile Height:" << getAttributeString(pRoot, "tileheight") << "\n";
46 m_tileSize = getAttributeInt(pRoot, "tilewidth");
47 m_width = getAttributeInt(pRoot, "width");
48 m_height = getAttributeInt(pRoot, "height");
49
50 // create the level object
51 Level* pLevel = new Level();
52 // we must parse the textures needed for this level, which have been added to properties
53 for(XMLElement* e = pRoot->FirstChildElement(); e != NULL; e = e->NextSiblingElement())
54 {
55 //we know that properties is the first child of the root
56 if(isElementNamed(e, "properties")) {
57 for(auto prop = e->FirstChildElement(); prop; prop = prop->NextSiblingElement()) {
58 if(isElementNamed(prop, "property")) { parseTextures(prop); }
59 }
60 }
61 // we must now parse the tilesets
62 else if(isElementNamed(e, "tileset")) {
63 parseTilesets(e, pLevel->getTilesets());
64 }
65 // parse any object layers
66 else if(isElementNamed(e, "objectgroup") || isElementNamed(e, "layer")) {
67 if(isElementNamed(e->FirstChildElement(), "object")) {
68 parseObjectLayer(e, pLevel->getLayers(), pLevel);
69 }
70 else if(isElementNamed(e->FirstChildElement(), "data") ||
71 (e->FirstChildElement()->NextSiblingElement() != 0 &&
72 e->FirstChildElement()->NextSiblingElement()->Value() == std::string("data"))) {
73 parseTileLayer(e, pLevel->getLayers(), pLevel->getTilesets(), pLevel->getCollisionLayers());
74 }
75 }
76 }
77 return pLevel;
78}

86void LevelParser::parseTilesets(XMLElement* pTilesetRoot, std::vector<Tileset>* pTilesets)
87{
88 std::string assetsTag = "assets/";
89 // first add the tileset to texture manager
90 std::cout << "adding texture " << pTilesetRoot->FirstChildElement()->Attribute("source") << " with ID " << pTilesetRoot->Attribute("name") << std::endl;
91 TheTextureManager::Instance()->load(assetsTag.append(pTilesetRoot->FirstChildElement()->Attribute("source")), pTilesetRoot->Attribute("name"));
92
93 // create a tileset object
94 Tileset tileset;
95 pTilesetRoot->FirstChildElement()->QueryIntAttribute("width", &tileset.width);
96 pTilesetRoot->FirstChildElement()->QueryIntAttribute("height", &tileset.height);
97 pTilesetRoot->QueryIntAttribute("firstgid", &tileset.firstGridID);
98 pTilesetRoot->QueryIntAttribute("tilewidth", &tileset.tileWidth);
99 pTilesetRoot->QueryIntAttribute("tileheight", &tileset.tileHeight);
100 pTilesetRoot->QueryIntAttribute("spacing", &tileset.spacing);
101 pTilesetRoot->QueryIntAttribute("margin", &tileset.margin);
102 tileset.name = pTilesetRoot->Attribute("name");
103 tileset.numColumns = tileset.width / (tileset.tileWidth + tileset.spacing);
104 pTilesets->push_back(tileset);
105}

107void LevelParser::parseObjectLayer(tinyxml2::XMLElement* pObjectElement, std::vector<Layer*>* pLayers, Level* pLevel)
108{
109 ObjectLayer* pObjectLayer = new ObjectLayer();
110 // 提前定义默认值
111 int x = 0, y = 0, width = 0, height = 0, numFrames = 1, callbackID = 0, animSpeed = 4;
112 std::string type, textureID;
113 // 遍历所有对象元素
114 for (auto e = pObjectElement->FirstChildElement(); e; e = e->NextSiblingElement())
115 {
116 if (!isElementNamed(e, "object")) continue;
117 // 重置默认值
118 x = getAttributeInt(e, "x");
119 y = getAttributeInt(e, "y");
120 type = getAttributeString(e, "type");
121 width = height = 0;
122 numFrames = 1;
123 callbackID = 0;
124 animSpeed = 4;
125 textureID.clear();
126 // 创建游戏对象
127 GameObject* pGameObject = TheGameObjectFactory::Instance()->create(type);
128 if (!pGameObject) {
129 std::cerr << "Failed to create game object of type: " << type << "\n";
130 continue;
131 }
132 // 解析属性 - 提取为单独函数以减少嵌套
133 parseObjectProperties(e, width, height, numFrames, textureID, callbackID, animSpeed);
134 // 加载游戏对象
135 pGameObject->load(std::make_unique<LoaderParams>(x, y, width, height, textureID, numFrames, callbackID, animSpeed));
136 // 设置玩家引用（如果适用）
137 if (type == "Player") {
138 pLevel->setPlayer(dynamic_cast<Player*>(pGameObject));
139 }
140 pObjectLayer->getGameObjects()->push_back(std::unique_ptr<GameObject>(pGameObject));
141 }
142 pLayers->push_back(pObjectLayer);
143}

145void LevelParser::parseObjectProperties(tinyxml2::XMLElement* objectElement, int& width, int& height, int& numFrames, std::string& textureID, int& callbackID, int& animSpeed)
146{
147 // 属性名到变量的映射（仅适用于整型）
148 std::unordered_map<std::string, int*> intProps = {
149 {"numFrames", &numFrames},
150 {"textureHeight", &height},
151 {"textureWidth", &width},
152 {"callbackID", &callbackID},
153 {"animSpeed", &animSpeed}
154 };
155 for (auto properties = objectElement->FirstChildElement(); properties; properties = properties->NextSiblingElement())
156 {
157 if (!isElementNamed(properties, "properties")) continue;
158 for (auto property = properties->FirstChildElement(); property; property = property->NextSiblingElement())
159 {
160 if (!isElementNamed(property, "property")) continue;
161
162 std::string propName = getAttributeString(property, "name");
163 if (propName == "textureID") {
164 textureID = getAttributeString(property, "value");
165 } else if (auto it = intProps.find(propName); it != intProps.end()) {
166 *(it->second) = getAttributeInt(property, "value");
167 }
168 }
169 }
170}

172void LevelParser::parseTileLayer(XMLElement* pTileElement, std::vector<Layer*> *pLayers,
173 const std::vector<Tileset>* pTilesets, std::vector<TileLayer*> *pCollisionLayers)
174{
175 TileLayer* pTileLayer = new TileLayer(m_tileSize, *pTilesets);
176 bool collidable = false;
177 // tile data
178 std::vector<std::vector<int>> data;
179 std::string decodedIDs;
180 XMLElement* pDataNode;
181 for(XMLElement* e = pTileElement->FirstChildElement(); e != NULL; e = e->NextSiblingElement())
182 {
183 if(e->Value() == std::string("properties"))
184 {
185 for(XMLElement* property = e->FirstChildElement(); property != NULL; property = property->NextSiblingElement())
186 {
187 if(property->Value() == std::string("property")) {
188 if(property->Attribute("name") == std::string("collidable")) {
189 collidable = true;
190 }
191 }
192 }
193 }
194 if(e->Value() == std::string("data")) {
195 pDataNode = e;
196 }
197 }
198
199 for(XMLNode* e = pDataNode->FirstChild(); e != NULL; e = e->NextSibling())
200 {
201 XMLText* text = e->ToText();
202 std::string t = text->Value();
203 t.erase(std::remove_if(t.begin(), t.end(), ::isspace), t.end());
204 decodedIDs += base64_decode(t);
205 }
206
207 // uncompress zlib compression
208 std::vector<uint32_t> ids(m_width * m_height);
209 uLongf sizeofids = ids.size() * sizeof(uint32_t);
210 int uncompress_result = uncompress(
211 (Bytef*)ids.data(),
212 &sizeofids,
213 (const Bytef*)decodedIDs.data(),
214 decodedIDs.size());
215 if(uncompress_result != Z_OK) {
216 std::cerr << "zlib uncompress failed! Error code: " << uncompress_result << "\n";
217 // 可能原因：数据不是 zlib 压缩、损坏、buffer 太小等
218 }
219 if (sizeofids != ids.size() * sizeof(uint32_t)) {
220 std::cerr << "Decompressed size mismatch!\n";
221 }
222
223 std::vector<int> layerRow(m_width);
224 for(int j = 0; j < m_height; j++) {
225 data.push_back(layerRow);
226 }
227
228 for(int rows = 0; rows < m_height; rows++) {
229 for(int cols = 0; cols < m_width; cols++) {
230 data[rows][cols] = ids[rows * m_width + cols];
231 }
232 }
233
234 pTileLayer->setTileIDs(data);
235 pTileLayer->setMapHeight(m_height);
236 pTileLayer->setMapWidth(m_width);
237 //std::cout << "Map hight-width:" << m_width << "," << m_height << "!\n";
238
239 if(collidable) {
240 pCollisionLayers->push_back(pTileLayer);
241 }
242 pLayers->push_back(pTileLayer);
243}
