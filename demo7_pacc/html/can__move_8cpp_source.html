15bool canMove(
16 entt::registry &reg,
17 const MazeState &maze,
18 const entt::entity e,
19 const Pos pos,
20 const Dir desiredDir
21) {
22 const Pos desiredPos = pos + toPos(desiredDir);
23
24 if (maze.outOfRange(desiredPos)) {
25 // The tunnel is the only place where playable space meets the edge of the map.
26 // The movement system handles the tunnel
27 return desiredDir == Dir::left || desiredDir == Dir::right;
28 }
29
30 // just like we assumed the position of the tunnel, we're assuming the
31 // orientation of the door to the ghost house. Once again, making the door
32 // into an entity is probably a more robust but more complex solution.
33 // I guess this is one of those "do as I say, not as I do" situations!
34 const Tile desiredTile = maze[desiredPos];
35 if (desiredTile == Tile::door) {
36 if (reg.has<EnterHouse>(e) && desiredDir == Dir::down) {
37 return true;
38 } else if (reg.has<LeaveHouse>(e) && desiredDir == Dir::up) {
39 return true;
40 }
41 } else if (desiredTile != Tile::wall) {
42 return true;
43 }
44
45 return false;
46}
