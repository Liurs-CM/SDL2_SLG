67void InputHandler::update()
68{
69 m_keysPressedThisFrame.clear();
70 SDL_Event event;
71 while(SDL_PollEvent(&event))
72 {
73 switch(event.type)
74 {
75 case SDL_QUIT:
76 TheGame::Instance()->quit();
77 break;
78
79 case SDL_JOYAXISMOTION:
80 onJoystickAxisMove(event);
81 break;
82
83 case SDL_JOYBUTTONDOWN:
84 onJoystickButtonDown(event);
85 break;
86
87 case SDL_JOYBUTTONUP:
88 onJoystickButtonUp(event);
89 break;
90
91 case SDL_MOUSEMOTION:
92 onMouseMove(event);
93 break;
94
95 case SDL_MOUSEBUTTONDOWN:
96 onMouseButtonDown(event);
97 break;
98
99 case SDL_MOUSEBUTTONUP:
100 onMouseButtonUp(event);
101 break;
102
103 case SDL_KEYDOWN:
104 onKeyDown();
105 if (event.key.repeat == 0)
106 {
107 SDL_Scancode sc = static_cast<SDL_Scancode>(event.key.keysym.scancode);
108 m_keysPressedThisFrame.insert(sc);
109 }
110 break;
111
112 case SDL_KEYUP:
113 onKeyUp();
114 break;
115
116 default:
117 break;
118 }
119 }
120}
