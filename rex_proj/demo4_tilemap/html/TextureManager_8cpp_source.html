98void TextureManager::drawText(int x, int y, const char *text, uint32_t color)
99{
100 if (!text) return;
101 int ox = x;
102 std::vector<SDL_Point> points;
103 for (const char *p = text; *p; p++) {
104 unsigned char ch = (unsigned char)*p;
105 if (ch == '\n') {
106 x = ox;
107 y += 10; // 8 pixels high + 2 pixels line spacing
108 continue;
109 }
110 if (ch < 32 || ch > 126) continue;
111 const unsigned char *glyph = _gamelib_font8x8[ch - 32];
112 for (int row = 0; row < 8; row++) {
113 unsigned char bits = glyph[row];
114 for (int col = 0; col < 8; col++) {
115 if (bits & (0x80 >> col)) {
116 points.push_back({ x + col, y + row});
117 }
118 }
119 }
120 const int pt_count = points.size();
121 if( pt_count > 0) {
122 Uint8 r = (color >> 16) & 0xFF;
123 Uint8 g = (color >> 8) & 0xFF;
124 Uint8 b = color & 0xFF;
125 SDL_Renderer* pRenderer = RenderContext::get();
126 SDL_SetRenderDrawColor(pRenderer, r, g, b, 255);
127 SDL_RenderDrawPoints(pRenderer, points.data(), pt_count);
128 }
129 x += 8;
130 }
131}

133void TextureManager::drawTextScale(int x, int y, const char *text, uint32_t color, int scale)
134{
135 if (!text || scale <= 0) return;
136 std::vector<SDL_Rect> rects;
137 int ox = x;
138 for (const char *p = text; *p; p++) {
139 unsigned char ch = (unsigned char)*p;
140 if (ch == '\n') {
141 x = ox;
142 y += (8 + 2) * scale;
143 continue;
144 }
145 if (ch < 32 || ch > 126) continue;
146 const unsigned char *glyph = _gamelib_font8x8[ch - 32];
147 for (int row = 0; row < 8; row++) {
148 unsigned char bits = glyph[row];
149 for (int col = 0; col < 8; col++) {
150 if (bits & (0x80 >> col)) {
151 rects.push_back({ x + col * scale, y + row * scale, scale, scale });
152 }
153 }
154 }
155 const int rect_count = rects.size();
156 if (rect_count > 0) {
157 Uint8 r = (color >> 16) & 0xFF;
158 Uint8 g = (color >> 8) & 0xFF;
159 Uint8 b = color & 0xFF;
160 SDL_Renderer* pRenderer = RenderContext::get();
161 SDL_SetRenderDrawColor(pRenderer, r, g, b, 255);
162 SDL_RenderFillRects(pRenderer, rects.data(), rect_count);
163 }
164 x += 8 * scale;
165 }

180void TextureManager::drawCircle(int cx, int cy, int radius, bool Fill, uint32_t color)
181{
182 if(radius <= 0) return;
183 if(Fill) {
184 // 动态计算分段数：保证每段弧长约 ≤ 2 像素
185 const int segments = std::max(8, static_cast<int>(M_PI * radius / 2.0f));
186 // 分配顶点（中心 + 圆周）
187 const int numVertices = segments + 2; // 中心 + segments+1 个圆周点
188 const int numIndices = segments * 3; // 每个三角形3个索引
189 std::vector<SDL_Vertex> vertices(numVertices);
190 std::vector<int> indices(numIndices);
191 // 中心点
192 vertices[0].position = { static_cast<float>(cx), static_cast<float>(cy) };
193 Uint8 r = (color >> 16) & 0xFF;
194 Uint8 g = (color >> 8) & 0xFF;
195 Uint8 b = color & 0xFF;
196 vertices[0].color = { r, g, b, 255 };
197 // 圆周点（闭合扇形）
198 const float step = 2.0f * static_cast<float>(M_PI) / segments;
199 for (int i = 0; i <= segments; ++i) {
200 float angle = i * step;
201 vertices[i + 1].position = {
202 static_cast<float>(cx + radius * std::cos(angle)),
203 static_cast<float>(cy + radius * std::sin(angle))
204 };
205 vertices[i + 1].color = { r, g, b, 255 };
206 }
207 // 构建索引：每个三角形由 (0, i+1, i+2) 构成，共 segments 个三角形
208 // 总索引数 = segments * 3
209 for (int i = 0; i < segments; ++i) {
210 indices[i * 3 + 0] = 0; // 中心
211 indices[i * 3 + 1] = i + 1; // 当前圆周点
212 indices[i * 3 + 2] = i + 2; // 下一个圆周点（自动闭合）
213 }
214 // 单次 GPU 调用绘制整个圆
215 SDL_RenderGeometry(RenderContext::get(), nullptr, vertices.data(), numVertices, indices.data(), numIndices);
216 }
217
218 else {
219 const int segments = std::max(16, static_cast<int>(M_PI * radius / 1.5f));
220 const int numPoints = segments + 1; // 中心 + segments+1 个圆周点
221 std::vector<SDL_Point> points(numPoints);
222 const float step = 2.0f * static_cast<float>(M_PI) / segments;
223 for (int i = 0; i <= segments; ++i) {
224 float angle = i * step;
225 points[i] = {
226 static_cast<int>(cx + radius * std::cos(angle)),
227 static_cast<int>(cy + radius * std::sin(angle))
228 };
229 }
230 Uint8 r = (color >> 16) & 0xFF;
231 Uint8 g = (color >> 8) & 0xFF;
232 Uint8 b = color & 0xFF;
233 SDL_Renderer* pRenderer = RenderContext::get();
234 SDL_SetRenderDrawColor(pRenderer, r, g, b, 255);
235 SDL_RenderDrawLines(pRenderer, points.data(), numPoints);
236 }
237}
