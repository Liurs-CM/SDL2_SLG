187 template<typename Component>
188 const pool_handler<Component> & assure() const {
189 static_assert(std::is_same_v<Component, std::decay_t<Component>>);
190
191 if constexpr(has_type_index_v<Component>) {
192 const auto index = type_index<Component>::value();
193
194 if(!(index < pools.size())) {
195 pools.resize(index+1);
196 }
197
198 if(auto &&pdata = pools[index]; !pdata.pool) {
199 pdata.type_id = type_info<Component>::id();
200 pdata.pool.reset(new pool_handler<Component>());
201 pdata.remove = [](sparse_set<entity_type> &cpool, basic_registry &owner, const entity_type entt) {
202 static_cast<pool_handler<Component> &>(cpool).remove(owner, entt);
203 };
204 }
205
206 return static_cast<pool_handler<Component> &>(*pools[index].pool);
207 } else {
208 sparse_set<entity_type> *candidate{nullptr};
209
210 if(auto it = std::find_if(pools.begin(), pools.end(), [id = type_info<Component>::id()](const auto &pdata) { return id == pdata.type_id; }); it == pools.cend()) {
211 candidate = pools.emplace_back(pool_data{
212 type_info<Component>::id(),
213 std::unique_ptr<sparse_set<entity_type>>{new pool_handler<Component>()},
214 [](sparse_set<entity_type> &cpool, basic_registry &owner, const entity_type entt) {
215 static_cast<pool_handler<Component> &>(cpool).remove(owner, entt);
216 }
217 }).pool.get();
218 } else {
219 candidate = it->pool.get();
220 }
221
222 return static_cast<pool_handler<Component> &>(*candidate);
223 }

480 entity_type create() {
481 entity_type entt;
482
483 if(destroyed == null) {
484 entt = entities.emplace_back(entity_type(entities.size()));
485 // traits_type::entity_mask is reserved to allow for null identifiers
486 ENTT_ASSERT(to_integral(entt) < traits_type::entity_mask);
487 } else {
488 const auto curr = to_integral(destroyed);
489 const auto version = to_integral(entities[curr]) & (traits_type::version_mask << traits_type::entity_shift);
490 destroyed = entity_type{to_integral(entities[curr]) & traits_type::entity_mask};
491 entt = entities[curr] = entity_type{curr | version};
492 }
493
494 return entt;

508 entity_type create(const entity_type hint) {
509 ENTT_ASSERT(hint != null);
510 entity_type entt;
511
512 if(const auto req = (to_integral(hint) & traits_type::entity_mask); !(req < entities.size())) {
513 entities.reserve(req + 1);
514
515 for(auto pos = entities.size(); pos < req; ++pos) {
516 entities.emplace_back(destroyed);
517 destroyed = entity_type(pos);
518 }
519
520 entt = entities.emplace_back(hint);
521 } else if(const auto curr = (to_integral(entities[req]) & traits_type::entity_mask); req == curr) {
522 entt = create();
523 } else {
524 auto *it = &destroyed;
525 for(; (to_integral(*it) & traits_type::entity_mask) != req; it = &entities[to_integral(*it) & traits_type::entity_mask]);
526 *it = entity_type{curr | (to_integral(*it) & (traits_type::version_mask << traits_type::entity_shift))};
527 entt = entities[req] = hint;
528 }
529
530 return entt;

1408 entt::basic_group<Entity, exclude_t<Exclude...>, get_t<Get...>, Owned...> group(get_t<Get...>, exclude_t<Exclude...> = {}) {
1409 static_assert(sizeof...(Owned) + sizeof...(Get) > 0);
1410 static_assert(sizeof...(Owned) + sizeof...(Get) + sizeof...(Exclude) > 1);
1411
1412 using handler_type = group_handler<exclude_t<Exclude...>, get_t<std::decay_t<Get>...>, std::decay_t<Owned>...>;
1413
1414 const auto cpools = std::forward_as_tuple(assure<std::decay_t<Owned>>()..., assure<std::decay_t<Get>>()...);
1415 constexpr auto size = sizeof...(Owned) + sizeof...(Get) + sizeof...(Exclude);
1416 handler_type *handler = nullptr;
1417
1418 if(auto it = std::find_if(groups.cbegin(), groups.cend(), [size](const auto &gdata) {
1419 return gdata.size == size
1420 && (gdata.owned(type_info<std::decay_t<Owned>>::id()) && ...)
1421 && (gdata.get(type_info<std::decay_t<Get>>::id()) && ...)
1422 && (gdata.exclude(type_info<Exclude>::id()) && ...);
1423 }); it != groups.cend())
1424 {
1425 handler = static_cast<handler_type *>(it->group.get());
1426 }
1427
1428 if(!handler) {
1429 group_data candidate = {
1430 size,
1431 { new handler_type{}, [](void *instance) { delete static_cast<handler_type *>(instance); } },
1432 []([[maybe_unused]] const id_type ctype) ENTT_NOEXCEPT { return ((ctype == type_info<std::decay_t<Owned>>::id()) || ...); },
1433 []([[maybe_unused]] const id_type ctype) ENTT_NOEXCEPT { return ((ctype == type_info<std::decay_t<Get>>::id()) || ...); },
1434 []([[maybe_unused]] const id_type ctype) ENTT_NOEXCEPT { return ((ctype == type_info<Exclude>::id()) || ...); },
1435 };
1436
1437 handler = static_cast<handler_type *>(candidate.group.get());
1438
1439 const void *maybe_valid_if = nullptr;
1440 const void *discard_if = nullptr;
1441
1442 if constexpr(sizeof...(Owned) == 0) {
1443 groups.push_back(std::move(candidate));
1444 } else {
1445 ENTT_ASSERT(std::all_of(groups.cbegin(), groups.cend(), [size](const auto &gdata) {
1446 const auto overlapping = (0u + ... + gdata.owned(type_info<std::decay_t<Owned>>::id()));
1447 const auto sz = overlapping + (0u + ... + gdata.get(type_info<std::decay_t<Get>>::id())) + (0u + ... + gdata.exclude(type_info<Exclude>::id()));
1448 return !overlapping || ((sz == size) || (sz == gdata.size));
1449 }));
1450
1451 const auto next = std::find_if_not(groups.cbegin(), groups.cend(), [size](const auto &gdata) {
1452 return !(0u + ... + gdata.owned(type_info<std::decay_t<Owned>>::id())) || (size > (gdata.size));
1453 });
1454
1455 const auto prev = std::find_if(std::make_reverse_iterator(next), groups.crend(), [](const auto &gdata) {
1456 return (0u + ... + gdata.owned(type_info<std::decay_t<Owned>>::id()));
1457 });
1458
1459 maybe_valid_if = (next == groups.cend() ? maybe_valid_if : next->group.get());
1460 discard_if = (prev == groups.crend() ? discard_if : prev->group.get());
1461 groups.insert(next, std::move(candidate));
1462 }
1463
1464 ((std::get<pool_handler<std::decay_t<Owned>> &>(cpools).super = std::max(std::get<pool_handler<std::decay_t<Owned>> &>(cpools).super, size)), ...);
1465
1466 (on_construct<std::decay_t<Owned>>().before(maybe_valid_if).template connect<&handler_type::template maybe_valid_if<std::decay_t<Owned>>>(*handler), ...);
1467 (on_construct<std::decay_t<Get>>().before(maybe_valid_if).template connect<&handler_type::template maybe_valid_if<std::decay_t<Get>>>(*handler), ...);
1468 (on_destroy<Exclude>().before(maybe_valid_if).template connect<&handler_type::template maybe_valid_if<Exclude>>(*handler), ...);
1469
1470 (on_destroy<std::decay_t<Owned>>().before(discard_if).template connect<&handler_type::discard_if>(*handler), ...);
1471 (on_destroy<std::decay_t<Get>>().before(discard_if).template connect<&handler_type::discard_if>(*handler), ...);
1472 (on_construct<Exclude>().before(discard_if).template connect<&handler_type::discard_if>(*handler), ...);
1473
1474 if constexpr(sizeof...(Owned) == 0) {
1475 for(const auto entity: view<Owned..., Get...>(entt::exclude<Exclude...>)) {
1476 handler->current.emplace(entity);
1477 }
1478 } else {
1479 // we cannot iterate backwards because we want to leave behind valid entities in case of owned types
1480 for(auto *first = std::get<0>(cpools).data(), *last = first + std::get<0>(cpools).size(); first != last; ++first) {
1481 handler->template maybe_valid_if<std::tuple_element_t<0, std::tuple<std::decay_t<Owned>...>>>(*this, *first);
1482 }
1483 }
1484 }
1485
1486 if constexpr(sizeof...(Owned) == 0) {
1487 return { handler->current, std::get<pool_handler<std::decay_t<Get>> &>(cpools)... };
1488 } else {
1489 return { std::get<0>(cpools).super, handler->current, std::get<pool_handler<std::decay_t<Owned>> &>(cpools)... , std::get<pool_handler<std::decay_t<Get>> &>(cpools)... };
1490 }
1491 }
