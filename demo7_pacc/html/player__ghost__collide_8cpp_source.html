35GhostCollision playerGhostCollide(entt::registry &reg) {
36 const auto players = reg.view<Player, Position, ActualDir>();
37 const auto ghosts = reg.view<Ghost, Position, ActualDir>();
38 for (const entt::entity p : players) {
39 const Pos playerPos = players.get<Position>(p).p;
40 const Dir playerDir = players.get<ActualDir>(p).d;
41 for (const entt::entity g : ghosts) {
42 const Pos ghostPos = ghosts.get<Position>(g).p;
43 const Dir ghostDir = ghosts.get<ActualDir>(g).d;
44 if (collide(playerPos, ghostPos, playerDir, ghostDir)) {
45 if (reg.has<ScaredMode>(g)) {
46 return {g, GhostCollision::Type::eat};
47 } else if (reg.has<EatenMode>(g)) {
48 continue;
49 }
50 return {g, GhostCollision::Type::lose};
51 }
52 }
53 }
54 return {entt::null, GhostCollision::Type::none};
