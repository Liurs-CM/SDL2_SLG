19void playerRender(entt::registry &reg, SDL::QuadWriter &writer, const int frame) {
20 const auto view = reg.view<Position, ActualDir, DesiredDir, PlayerSprite>();
21 for (const entt::entity e : view) {
22 const Pos pos = view.get<Position>(e).p * tileSize;
23 const Dir actualDir = view.get<ActualDir>(e).d;
24 const double angle = static_cast<double>(view.get<DesiredDir>(e).d) * 90.0;
25 writer.tilePos(pos + toPos(actualDir, frame), Pos{tileSize, tileSize}, angle);
26 writer.tileTex(view.get<PlayerSprite>(e).id + frame);
27 writer.render();
28 }
29}

31void ghostRender(entt::registry &reg, SDL::QuadWriter &writer, const int frame) {
32 const auto view = reg.view<Position, ActualDir, GhostSprite>();
33 for (const entt::entity e : view) {
34 const Pos pos = view.get<Position>(e).p * tileSize;
35 const Dir actualDir = view.get<ActualDir>(e).d;
36 writer.tilePos(pos + toPos(actualDir, frame), Pos{tileSize, tileSize});
37 const int dirOffset = (
38 actualDir == Dir::none ? 0 : static_cast<int>(actualDir)
39 );
40 const GhostSprite sprite = view.get<GhostSprite>(e);
41 if (reg.has<ChaseMode>(e) || reg.has<ScatterMode>(e)) {
42 writer.tileTex(sprite.id + dirOffset);
43 } else if (reg.has<ScaredMode>(e)) {
44 const int scaredTimer = reg.get<ScaredMode>(e).timer;
45 const int flash = (
46 scaredTimer <= ghostScaredFlashTime ? (frame / ghostScaredFlashRate) % 2 : 0
47 );
48 writer.tileTex(animera::SpriteID::scared_beg_ + flash);
49 } else if (reg.has<EatenMode>(e)) {
50 writer.tileTex(animera::SpriteID::eyes_beg_ + dirOffset);
51 }
52 writer.render();
53 }
54}

56void dotRender(SDL::QuadWriter &writer, const MazeState &maze) {
57 for (int y = 0; y != maze.height(); ++y) {
58 for (int x = 0; x != maze.width(); ++x) {
59 const Tile tile = maze[{x, y}];
60 writer.tilePos({x * tileSize, y * tileSize}, {tileSize, tileSize});
61 if (tile == Tile::dot) {
62 writer.tileTex(animera::SpriteID::dot);
63 } else if (tile == Tile::energizer) {
64 writer.tileTex(animera::SpriteID::energizer);
65 } else {
66 continue;
67 }
68 writer.render();
69 }
70 }
