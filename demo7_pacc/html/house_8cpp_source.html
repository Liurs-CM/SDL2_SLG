19void enterHouse(entt::registry &reg) {
20 const auto view = reg.view<EnterHouse, Position, HomePosition>();
21 for (const entt::entity e : view) {
22 if (view.get<Position>(e).p == view.get<HomePosition>(e).home) {
23 reg.remove<EnterHouse>(e);
24 reg.emplace<LeaveHouse>(e);
25 reg.remove<EatenMode>(e);
26 reg.emplace<ChaseMode>(e);
27 }
28 }
29}

31void leaveHouse(entt::registry &reg) {
32 auto view = reg.view<LeaveHouse, Position, Target>();
33 for (const entt::entity e : view) {
34 if (view.get<Position>(e).p == outsideHouse) {
35 reg.remove<LeaveHouse>(e);
36 } else {
37 // leaveHouse is called after the set*Target systems so this will
38 // overwrite the target position. It's a little bit hacky but it
39 // works here.
40
41 // We actually need to do this because ghosts can sometimes get stuck
42 // in the house. If their target is towards the bottom of the maze,
43 // they will move left and right without going up and out of the house
44 view.get<Target>(e).p = outsideHouse;
45 }
46 }
47}
