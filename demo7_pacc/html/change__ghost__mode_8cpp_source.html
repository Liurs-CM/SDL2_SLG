16void ghostScared(entt::registry &reg) {
17 const auto view = reg.view<Ghost>();
18 for (const entt::entity e : view) {
19 reg.remove_if_exists<ChaseMode, ScatterMode, ScaredMode>(e);
20 // Ghosts in EatenMode don't get scared
21 if (!reg.has<EatenMode>(e)) {
22 reg.emplace<ScaredMode>(e);
23 }
24 }
25}

27void ghostScaredTimeout(entt::registry &reg) {
28 auto view = reg.view<Ghost, ScaredMode>();
29 for (const entt::entity e : view) {
30 ScaredMode &scared = view.get<ScaredMode>(e);
31 --scared.timer;
32 if (scared.timer <= 0) {
33 // Adding and removing components from the entity that is currently
34 // returned by the view is OK
35 reg.remove<ScaredMode>(e);
36 reg.emplace<ChaseMode>(e);
37 }
38 }
39}

41void ghostEaten(entt::registry &reg, const entt::entity ghost) {
