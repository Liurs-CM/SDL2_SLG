57void Shape::initHealthBarSystem()
58{
59 // 创建 256x1 渐变纹理（红→黄→绿）
60 SDL_Surface* s = SDL_CreateRGBSurface(0, 256, 1, 32, 0, 0, 0, 0);
61 for (int x = 0; x < 256; ++x) {
62 float t = x / 255.0f;
63 Uint8 r = (t < 0.5f) ? 255 : (Uint8)(255 * (1 - (t - 0.5f) * 2));
64 Uint8 g = (t < 0.5f) ? (Uint8)(255 * t * 2) : 255;
65 Uint8 b = 0;
66 ((Uint32*)s->pixels)[x] = SDL_MapRGB(s->format, r, g, b);
67 }
68 gHealthGradient = SDL_CreateTextureFromSurface(pRenderer, s);
69 SDL_FreeSurface(s);
70}

73void Shape::drawHealthBar(
74 int x, int y, // 屏幕位置
75 int width, int height, // 血条尺寸
76 int currentHP, int maxHP) // 绝对血量值

78 SDL_Renderer* pRenderer = RenderContext::get();
79 const int SEG_SIZE = 100;
80 const int MAX_SEG = 10;
81
82 // 1. 计算段数（按绝对值）
83 int curSeg = std::min(currentHP / SEG_SIZE, MAX_SEG);
84 int maxSeg = std::min(maxHP / SEG_SIZE, MAX_SEG);
85 if (maxSeg == 0) maxSeg = 1;
86 if (currentHP > 0 && curSeg == 0) curSeg = 1;
87
88 float ratio = (float)curSeg / maxSeg;
89
90 // 2. 背景（可选）
91 SDL_SetRenderDrawColor(pRenderer, 127, 127, 127, 255);
92 SDL_Rect rect = {x, y, width, height};
93 SDL_RenderFillRect(pRenderer, &rect);
94
95 // 3. 渐变前景
96 if (ratio > 0.0f) {
97 SDL_Rect src = {0, 0, (int)(256 * ratio), 1};
98 SDL_Rect dst = {x, y, (int)(width * ratio), height};
99 SDL_RenderCopy(pRenderer, gHealthGradient, &src, &dst);
100 }
101
102 // 4. ★ 1像素黑边框 ★
103 SDL_SetRenderDrawColor(pRenderer, 0, 0, 0, 255);
104 SDL_RenderDrawRect(pRenderer, &rect);
105
106 // 5. ★ >1000 高亮（白边）★
107 if (maxHP >= 1000) {
108 SDL_SetRenderDrawColor(pRenderer, 255, 255, 255, 255);
109 rect = {x - 1, y - 1, width + 2, height + 2};
110 SDL_RenderDrawRect(pRenderer, &rect);
111 }
112
113 // 6. ★ 1像素黑点倒角 ★
114 SDL_SetRenderDrawColor(pRenderer, 127, 127, 127, 255);
115 SDL_RenderDrawPoint(pRenderer, x - 1, y - 1);
116 SDL_RenderDrawPoint(pRenderer, x - 1, y + height);
117 SDL_RenderDrawPoint(pRenderer, x + width, y - 1);
118 SDL_RenderDrawPoint(pRenderer, x + width, y + height);
119}
