18Dir readDir(const SDL_Scancode key) {
19 switch (key) {
20 case SDL_SCANCODE_W:
21 case SDL_SCANCODE_UP:
22 return Dir::up;
23 case SDL_SCANCODE_D:
24 case SDL_SCANCODE_RIGHT:
25 return Dir::right;
26 case SDL_SCANCODE_S:
27 case SDL_SCANCODE_DOWN:
28 return Dir::down;
29 case SDL_SCANCODE_A:
30 case SDL_SCANCODE_LEFT:
31 return Dir::left;
32 default:
33 return Dir::none;
34 }

39bool playerInput(entt::registry &reg, const SDL_Scancode key) {
40 const Dir dir = readDir(key);
41 if (dir == Dir::none) {
42 return false;
43 }
44 auto view = reg.view<Player, DesiredDir>();
45 for (const entt::entity e : view) {
46 view.get<DesiredDir>(e).d = dir;
47 }
48 return true;
