22int getScaleFactor() {
23 // Make the largest window possible with an integer scale factor
24 SDL_Rect bounds;
25 #if SDL_VERSION_ATLEAST(2, 0, 5)
26 SDL_CHECK(SDL_GetDisplayUsableBounds(0, &bounds));
27 #else
28 #warning SDL 2.0.5 or later is recommended
29 SDL_CHECK(SDL_GetDisplayBounds(0, &bounds));
30 #endif
31 const int scaleX = bounds.w / tilesPx.x;
32 const int scaleY = bounds.h / tilesPx.y;
33 return std::max(1, std::min(scaleX, scaleY));
34}

46void Application::run() {
47 const int scaleFactor = getScaleFactor();
48 std::cout << "Using scale factor: " << scaleFactor << '\n';
49
50 SDL::Window window{SDL_CHECK(SDL_CreateWindow(
51 "Pacman",
52 SDL_WINDOWPOS_CENTERED, SDL_WINDOWPOS_CENTERED,
53 tilesPx.x * scaleFactor, tilesPx.y * scaleFactor,
54 SDL_WINDOW_SHOWN
55 ))};
56
57 SDL::Renderer renderer{SDL_CHECK(SDL_CreateRenderer(
58 window.get(),
59 -1,
60 SDL_RENDERER_ACCELERATED | SDL_RENDERER_PRESENTVSYNC
61 ))};
62 SDL_CHECK(SDL_RenderSetLogicalSize(renderer.get(), tilesPx.x, tilesPx.y));
63
64 SDL::Texture maze = SDL::loadTexture(renderer.get(), animera::getTextureInfo());
65 SDL::QuadWriter writer{renderer.get(), maze.get()};
66 Game game;
67 game.init();
68
69 int frame = 0;
70 bool quit = false;
71 while (!quit) {
72 FrameCap sync{fps};
73
74 SDL_Event e;
75 while (SDL_PollEvent(&e)) {
76 if (e.type == SDL_QUIT) {
77 quit = true;
78 break;
79 } else if (e.type == SDL_KEYDOWN) {
80 game.input(e.key.keysym.scancode);
81 }
82 }
83
84 // Game::logic is called once for each tile
85 // Game::render is called for each pixel between tiles
86 if (frame % tileSize == 0) {
87 if (!game.logic()) {
88 quit = true;
89 }
90 }
91
92 SDL_CHECK(SDL_SetRenderDrawColor(renderer.get(), 0, 0, 0, 255));
93 SDL_CHECK(SDL_RenderClear(renderer.get()));
94 game.render(writer, frame % tileSize);
95 ++frame;
96 SDL_RenderPresent(renderer.get());
97 }
98}
