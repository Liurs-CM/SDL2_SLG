14// Tag types for the ghost modes. This is a better alternative to an enum.
15// Creating a struct for each mode means that we store data specific to
16// the mode. ScaredMode stores a timer. That timer only makes sense when
17// the ghost is in ScaredMode. I could have used a std::variant but I think
18// this is cleaner
19
20struct ChaseMode {};
21struct ScatterMode {};
22struct ScaredMode {
23 // This timer counts down the number of ticks that the ghost is scared for.
24 // Once the timer reaches zero, the ghost enters chase mode. This is handled
25 // by the change_ghost_mode systems

27};
